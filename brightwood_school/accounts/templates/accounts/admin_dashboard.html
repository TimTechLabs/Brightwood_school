from django.shortcuts import render, get_object_or_404 from django.db.models import Sum from accounts.models import Student from finance.models import FeePayment from examinations.models import Mark def dashboard_view(request): # Fetch search parameter or default to '12028' adm_no = request.GET.get('tq', '12028').strip() # Safely retrieve student object or raise a 404 student = get_object_or_404(Student, adm_number=adm_no) # Query fee records (ordered by most recent) fee_payments = FeePayment.objects.filter(student=student).order_by('-id') latest_fee = fee_payments.first() # Query marks with optimized foreign key prefetching marks = Mark.objects.filter(student=student) # Optional aggregate calculation for total fees paid total_fees_paid = fee_payments.aggregate(Sum('amount'))['amount__sum'] or 0.00 context = { 'student': student, 'fee_payments': fee_payments, 'latest_fee': latest_fee, 'total_fees_paid': total_fees_paid, 'marks': marks, } return render(request, 'dashboard.html', context)
